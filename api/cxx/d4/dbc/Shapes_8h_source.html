20 class BoundingBox : public OGREnvelope {
21 public:
22 // Expose the OGREnvelope::Merge method, so that the old function interface is preserved.
23 using OGREnvelope::Merge;
24
25 // Overload the method by adding parameter for EPSG in geographic coordinates.
26 // If the bounding boxes are in geographic coordinates, the overall longitude
27 // extent is assumed to be <= 180 degrees. If an antimeridian crossing is
28 // detected, the longitude coordinates are re-wrapped to the interval [0, 360).
29 void Merge(const BoundingBox& other, int epsg) {
30 double minx_global = std::min(MinX, other.MinX);
31 double maxx_global = std::max(MaxX, other.MaxX);
32
33 OGRSpatialReference bbox_srs;
34 bbox_srs.importFromEPSG(epsg);
35
36 // Check if antimeridian handling is necessary (i.e. geographic SRS AND crossing antimeridian).
37 // If not, use the method in the base class
38 if (!bbox_srs.IsGeographic() || ((maxx_global - minx_global) <= 180.0)) {
39 OGREnvelope::Merge(other);
40 return;
41 }
42
43 // Wrap the input angle (in degrees) to the interval [0, 360).
44 auto wrap360 = [](double x) {
45 auto y = std::fmod(x, 360.0);
46 if (y < 0.0) {
47 y += 360.0;
48 }
49 return y;
50 };
51
52 double MinX_wrap = wrap360(MinX);
53 double MaxX_wrap = wrap360(MaxX);
54
55 double other_MinX_wrap = wrap360(other.MinX);
56 double other_MaxX_wrap = wrap360(other.MaxX);
57
58
59 // In general, `MinX` is the east edge and `MaxX` is the west edge,
60 // unless the bounding box contained the antimeridian and the
61 // coordinates weren't "unwrapped" (such as if the bounding box
62 // was created using `Perimeter::getEnvelope()`). We assume this is
63 // the case if the difference between `MaxX` and `MinX` exceeds 180
64 // degrees. In this case, we can normalize the bounding box by
65 // swapping `MinX` and `MaxX` and re-wrapping the coordinates to the
66 // interval [0, 360).
67 if (MaxX - MinX > 180.0) {
68 std::swap(MinX_wrap, MaxX_wrap);
69 }
70 if (other.MaxX - other.MinX > 180.0) {
71 std::swap(other_MinX_wrap, other_MaxX_wrap);
72 }
73
74 MinX = std::min(MinX_wrap, other_MinX_wrap);
75 MaxX = std::max(MaxX_wrap, other_MaxX_wrap);
76
77 // merge y boundary
78 MinY = std::min(MinY, other.MinY);
79 MaxY = std::max(MaxY, other.MaxY);
80
81 }
82 };
