 11def exp_from_quantile(p, vp, bw=1.0):
 12    """
 13    Determine the rate parameter of the lifted exponential distribution from a
 14    quantile and its corresponding value.
 15
 16    Parameters
 17    ----------
 18    p : float | np.ndarray
 19        Quantile in [1-bw, 1).  For example, 0.5 for the median.
 20    vp : float | np.ndarray
 21        Value corresponding to the given quantile.  For example, the median
 22        value.
 23    bw : float, optional
 24        The portion of the distribution governed by an exponential distribution.
 25        In the context of the tone-rank algorithm, it's the chirp bandwidth
 26        normalized by the sample rate.  Values in interval (0, 1].
 27        See notes below.
 28
 29    Returns
 30    -------
 31    λ : float | np.ndarray
 32        The rate parameter of the lifted exponential distribution.
 33
 34    Notes
 35    -----
 36    The definition of the "lifted" exponential distribution is as follows:
 37        cdf(x) = (1 - bw) * u(x - x0) + bw * (1 - exp(-λ * x))
 38    where u(x) is the unit step function, x0 is some small value (e.g., the
 39    noise floor), and other parameters are as described above.
 40
 41    The idea here is that spectral power in the chirp band should follow an
 42    exponential distribution, while the values outside the chirp band will be
 43    smaller and follow some other distribution whose shape we don't care about
 44    and just model with a step function.
 45
 46    The formula is the same to solve for a value given the rate parameter.
 47    That is, you can also use this function to solve
 48        vp = exp_from_quantile(p, λ, bw)
 49    """
 50    p = np.asarray(p)
 51    if not np.all(p >= (1.0 - bw)):
 52        raise ValueError("invalid quantile for lifted exponential")
 53    vp = np.asarray(vp)
 54    return -np.log((1 - p) / bw) / vp

 61def get_spectral_mask(
 62    spectra,
 63    reference_quantile=0.5,
 64    nominal_false_positive_rate=0.0005,
 65    bandwidth=5 / 6,
 66):
 67    """
 68    Detect RFI in spectral domain data using lifted exponential model.
 69
 70    Uses a statistical model to identify spectral samples that are anomalously
 71    loud compared to the expected distribution of the signal. The method assumes
 72    that the power spectrum follows a lifted exponential distribution and uses
 73    a quantile of the data to estimate the distribution parameters.
 74
 75    Parameters
 76    ----------
 77    spectra : np.ndarray
 78        Complex spectral data, arbitrary shape.
 79    reference_quantile : float, optional
 80        Quantile (in [1-bandwidth, 1)) used to estimate the rate parameter
 81        of the lifted exponential distribution. Default is 0.5 (median).
 82    nominal_false_positive_rate : float, optional
 83        Target false positive rate for RFI detection, in (0, 1).
 84        Lower values result in more conservative detection (fewer false alarms
 85        but potentially missed RFI).
 86    bandwidth : float, optional
 87        Fraction of samples in (0, 1] assumed to follow the exponential
 88        distribution (as opposed to being noise, filter roll-off, etc).
 89
 90    Returns
 91    -------
 92    mask : np.ndarray[bool]
 93        Boolean mask marking detected RFI samples, same shape as spectra.
 94        True indicates RFI detection (sample exceeds threshold).
 95    isr : float
 96        Interference-to-signal ratio: ratio of total power in detected RFI
 97        samples to total power in clean samples.
 98    λ : float
 99        Estimated rate parameter of the lifted exponential distribution.
100
101    Notes
102    -----
103    The detection threshold is determined by inverting the cumulative
104    distribution function of the lifted exponential model at the desired
105    false positive rate. See exp_from_quantile for details on the
106    statistical model.
107    """
108    n = np.prod(spectra.shape)
109    power_spectra = abs2(spectra)
110    power_spectra.shape = (n,)
111    # partition seems to be a little faster than quantile, I guess because the
112    # latter does a full sort and interpolates?  For large n the difference is
113    # small.
114    k = round(reference_quantile * n)
115    kth_power = np.partition(power_spectra, k)[k]
116    if kth_power == 0.0:
117        return np.zeros(spectra.shape, dtype=bool), 0.0, 0.0
118    # Estimate the parameter of a lifted exponential distribution.
119    λ = exp_from_quantile(reference_quantile, kth_power, bandwidth)
120    # Determine threshold.  Since we've estimated a statistical model, we can
121    # achieve a specified nominal false positive rate by thresholding the
122    # nominal CDF at one minus that value.
123    q = 1.0 - nominal_false_positive_rate
124    too_loud = exp_from_quantile(q, λ, bandwidth)
125    # Use threshold to generate mask.
126    mask = power_spectra >= too_loud
127    # Compute interference-to-signal ratio now that we've labeled the data.
128    # Since masked values are presumably the sum of signal and interference,
129    # we'll include an arithmetic correction using the stats model.
130    mean_power = 1 / λ
131    masked_sig_power = np.sum(mask) * mean_power
132    rfi_power = np.sum(power_spectra[mask]) - masked_sig_power
133    sig_power = np.sum(power_spectra[~mask]) + masked_sig_power
134    isr = rfi_power / sig_power
135    # Unflatten data.
136    mask.shape = spectra.shape
137    return mask, isr, λ

140def fill_missing(z, fd, t, mask_replace, valid_rows, noise, interpolate=True,
141                 fill_value="noise"):
142    """
143    Fill missing/RFI-contaminated samples in spectral domain data.
144
145    Parameters
146    ----------
147    z : np.ndarray
148        Complex spectral data to fill, shape (m, n)
149    fd : float
150        Doppler centroid frequency in Hz.  It is not scaled for each frequency
151        bin in `z`.  If wideband processing is needed, just baseband `z` before
152        calling this function and set `fd=0`.
153    t : np.ndarray
154        Pulse times in seconds since epoch, length m
155    mask_replace : np.ndarray
156        Boolean mask marking samples to replace, shape (m, n)
157    valid_rows : np.ndarray
158        Boolean mask marking valid rows, shape (m,)
159    noise : np.ndarray
160        Random noise samples in 1D array for fallback replacement.  Values may
161        be used multiple times if length is less numpy.prod((m, n)).
162    interpolate : bool, optional
163        If True, attempt linear/nearest-neighbor interpolation. If False,
164        replace directly with fill_value. Default is True.
165    fill_value : str, optional
166        Fallback strategy when interpolation fails or is disabled.
167        "noise": use provided noise vector (default)
168        "zero": use zero
169
170    Returns
171    -------
172    zout : np.ndarray
173        Filled data, same shape as z
174    """
175    m, n = z.shape
176    if len(t) != m:
177        raise ValueError(f"expected len(t)=={m} got {len(t)}")
178    if mask_replace.shape != (m, n):
179        raise ValueError(f"{mask_replace.shape=} does not match {z.shape=}")
180    if valid_rows.shape != (m,):
181        raise ValueError(f"{valid_rows.shape=} does not match {z.shape[0]=}")
182    if interpolate and m < 2:
183        log.warning(f"Disabling interpolation for block of {m} rows "
184            "because there are no neighbors to use for interpolation.")
185        interpolate = False
186
187    # Deramp Doppler.
188    deramp = np.exp(-1j * 2 * np.pi * fd * t).astype(z.dtype)
189    zout = deramp[:, None] * z
190
191    # Exclude RFI samples from being used as interpolation sources.
192    mask_valid_clean = valid_rows[:, None] & ~mask_replace
193
194    for i in range(m):
195        # copy for modification
196        cols_need_replacement = mask_replace[i, :].copy()
197
198        if interpolate:
199            # Previous and next pulse, with reflection boundary condition.
200            iprev, inext = i - 1, i + 1
201            if i == 0:
202                iprev = i + 1
203            if i == m - 1:
204                inext = i - 1
205
206            # Non-uniform time sampling, so let's weight closer samples more.
207            # NOTE abs() since we might've reflected.
208            dt_prev = abs(t[i] - t[iprev])
209            dt_next = abs(t[inext] - t[i])
210            w_prev = dt_next / (dt_prev + dt_next)
211            w_next = dt_prev / (dt_prev + dt_next)
212
213            # Four cases for replacement:
214            # 1. prev and next both valid -> lerp between them
215            j = np.where(mask_valid_clean[iprev, :] & mask_valid_clean[inext, :]
216                & cols_need_replacement)[0]
217            zout[i, j] = w_prev * zout[iprev, j] + w_next * zout[inext, j]
218            cols_need_replacement[j] = False
219
220            # 2. only prev valid. use it
221            j = np.where(mask_valid_clean[iprev, :] & cols_need_replacement)[0]
222            zout[i, j] = zout[iprev, j]
223            cols_need_replacement[j] = False
224
225            # 3. only next valid. use it
226            j = np.where(mask_valid_clean[inext, :] & cols_need_replacement)[0]
227            zout[i, j] = zout[inext, j]
228            cols_need_replacement[j] = False
229
230        # 4. Fallback for remaining samples (or all if not interpolating)
231        j = np.where(cols_need_replacement)[0]
232        if len(j) > 0:
233            if fill_value == "zero":
234                zout[i, j] = 0.0
235            elif fill_value == "noise":
236                noise_idx = ((i * n) + j) % len(noise)
237                zout[i, j] = noise[noise_idx]
238            else:
239                raise ValueError(f"Invalid fill_value: {fill_value}")
240
241    # Put Doppler back on.
242    zout *= deramp[:, None].conj()
243    return zout

250def remove_loud_tones(
251    z,
252    t,
253    r,
254    swaths,
255    doppler,
256    block_dims=(512, 1024),
257    reference_quantile=0.5,
258    nominal_false_positive_rate=0.0005,
259    bandwidth=5 / 6,
260    detect_only=False,
261    zout=None,
262    interpolate=True,
263    fill_value="noise",
264    max_gap_fraction=0.5,
265):
266    """
267    Detect and optionally mitigate narrowband RFI using spectral rank method.
268
269    Processes raw data in overlapping blocks using a Short-Time Fourier Transform
270    (STFT) approach. RFI is detected in the spectral domain using a lifted
271    exponential statistical model. Detected RFI samples are replaced using
272    temporal interpolation or fill values.
273
274    Parameters
275    ----------
276    z : np.ndarray[complex64]
277        Raw data, shape (num_pulses, num_range_bins).
278    t : np.ndarray[float64]
279        Pulse times in seconds since orbit/grid epoch, length num_pulses.
280    r : isce3.core.Linspace
281        Slant range to each sample in meters.
282    swaths : np.ndarray[int]
283        Valid subswath samples, shape (num_subswaths, num_pulses, 2).
284        Last dimension contains [start, stop) indices of each subswath.
285    doppler : isce3.core.LUT2d
286        Raw data Doppler centroid look-up table in Hz. Must be valid over
287        entire grid.
288    block_dims : tuple[int, int], optional
289        Processing block size (azimuth, range). Default is (512, 1024).
290    reference_quantile : float, optional
291        Quantile used to estimate the rate parameter of the lifted exponential
292        distribution, in [1-bandwidth, 1). Default is 0.5 (median).
293    nominal_false_positive_rate : float, optional
294        Target false positive rate for RFI detection, in (0, 1].
295    bandwidth : float, optional
296        Ratio of chirp bandwidth to sample rate, in (0, 1].
297    detect_only : bool, optional
298        If True, only detect RFI without mitigation. If False, replace
299        detected RFI samples.
300    zout : np.ndarray[complex64], optional
301        Output buffer for mitigated data. Must have same shape as z.
302        If None, z is modified in-place.  In that case some rows will be cleaned
303        twice if block_dims[0] doesn't divide evenly into z.shape[0].
304    interpolate : bool, optional
305        If True, attempt linear/nearest-neighbor interpolation for RFI samples.
306        If False, replace directly with fill_value.
307    fill_value : str, optional
308        Fallback value when interpolation fails or is disabled.
309        "noise": use random Gaussian noise (default)
310        "zero": use zero
311    max_gap_fraction : float, optional
312        Max portion of a row that can be masked by a TX gap before its
313        spectrum is considered invalid (0.0: any gap invalidates row,
314        1.0: ignore gap mask).
315
316    Returns
317    -------
318    block_times : np.ndarray[float]
319        Azimuth time at center of each azimuth block in seconds since epoch,
320        length num_az_blocks.
321    block_ranges : np.ndarray[float]
322        Slant range at center of each range block in meters,
323        length num_range_blocks.
324    f : np.ndarray[float]
325        Normalized frequency axis for hits array, length block_dims[1].
326        Units are cycles per sample.
327    means : np.ndarray[float]
328        Estimated mean signal power per block from lifted exponential model,
329        shape (num_az_blocks, num_range_blocks). Equal to 1/λ where λ is the
330        rate parameter.
331    isr : np.ndarray[float]
332        Interference-to-signal ratio per block,
333        shape (num_az_blocks, num_range_blocks).
334    hits : np.ndarray[float]
335        Fraction of pulses with detected RFI at each frequency bin,
336        shape (num_az_blocks, num_range_blocks, block_dims[1]).
337        Values range from 0 (no RFI detected) to 1 (RFI detected in all pulses).
338
339    Notes
340    -----
341    The algorithm processes data in overlapping blocks using COLA (Constant
342    Overlap-Add) windowing in range. Each block is transformed to the spectral
343    domain where RFI tones appear as anomalously loud samples. Detection uses
344    the lifted exponential model (see get_spectral_mask). When mitigation is
345    enabled, detected samples are replaced using temporal interpolation from
346    adjacent clean pulses, with fallback to random noise or zeros.
347    """
348    # Check inputs
349    if not (z.ndim == len(block_dims) == 2):
350        raise ValueError("Data and block_dims must be 2-dimensional")
351    block_dims = (block_dims[0], block_dims[1])  # copy
352    for idim in (0, 1):
353        if block_dims[idim] > z.shape[idim]:
354            log.warning(f"Truncating block_dims[{idim}] to z.shape[{idim}]")
355            block_dims[idim] = z.shape[idim]
356    if not (0 < nominal_false_positive_rate <= 1):
357        raise ValueError(
358            "nominal_false_positive rate must be normalized to interval (0, 1]"
359        )
360    if not (0 < bandwidth <= 1):
361        raise ValueError("bandwidth must be normalized to interval (0, 1]")
362    if not ((1 - bandwidth) < reference_quantile < 1):
363        raise ValueError(
364            "reference_quantile must fall in the normalized signal "
365            f"distribution ({1 - bandwidth}, 1)"
366        )
367    if zout is None:
368        zout = z
369    if (not detect_only) and (zout.shape != z.shape):
370        raise ValueError("output shape must match input shape")
371
372    mask_valid = np.zeros((block_dims[0], z.shape[1]), bool)
373
374    slices_windows = list(cola_windows(z.shape[1], block_dims[1]))
375    num_range_blocks = len(slices_windows)
376    az_blocks = list(cpi_slice_gen(z.shape[0], block_dims[0]))
377    num_az_blocks = len(az_blocks)
378    zbshape = (block_dims[0], num_range_blocks, block_dims[1])
379    z_block = np.zeros(zbshape, z.dtype)
380
381    f = fftshift(fftfreq(block_dims[1]))
382    meta_shape = (num_az_blocks, num_range_blocks)
383    isr = np.zeros(meta_shape, dtype=np.float32)
384    means = np.zeros(meta_shape, dtype=np.float32)
385    hits = np.zeros(meta_shape + (block_dims[1],), dtype=np.float32)
386
387    block_times = np.zeros(num_az_blocks)
388    block_ranges = np.zeros(num_range_blocks)
389    for j, (cols, _) in enumerate(slices_windows):
390        # TODO could weight by window
391        block_ranges[j] = r[(cols.start + cols.stop) // 2]
392
393    # Generate a pool of noise with σ=1 once, and we'll scale it for each
394    # block as needed.  We'll generate a random offset for each block to
395    # increase entropy without having to regenerate noise for each block.
396    max_offset = num_az_blocks * num_range_blocks
397    num_noise = max_offset + np.prod(block_dims)
398    std_noise = circular_gaussian_noise(num_noise)
399
400    for iblock, rows in enumerate(az_blocks):
401        block_results = []
402        # calculate azimuth time of block
403        block_times[iblock] = t[rows].mean()
404        pulse_times = t[rows]
405        # populate valid data mask
406        mask_valid[...] = False
407        for i, i_pulse in enumerate(range(rows.start, rows.stop)):
408            for start, end in swaths[:, i_pulse, :]:
409                mask_valid[i, start:end] = True
410        # apply window to each range block
411        for j, (cols, window) in enumerate(slices_windows):
412            nw = len(window)
413            z_block[:, j, :nw] = window[None, :] * z[rows, cols]
414        # Range STFT.  Use consistent FFT size even for edges where window
415        # may be shorter so that frequency metadata are consistent.
416        spectra = fft(z_block, n=block_dims[1], axis=2)
417        # filter RFI
418        for j in range(z_block.shape[1]):
419            mask_replace, isr[iblock,j], λ = get_spectral_mask(
420                spectra[:, j, :],
421                reference_quantile,
422                nominal_false_positive_rate,
423                bandwidth,
424            )
425            if not detect_only:
426                σ = np.sqrt(0.5 / λ) if λ > 0.0 else 0.0
427                noise = σ * std_noise[np.random.randint(0, max_offset):]
428                fd = doppler.eval(block_times[iblock], block_ranges[j])
429                cols, window = slices_windows[j]
430                valid_rows = ((~mask_valid[:, cols]).mean(axis=1)
431                    <= max_gap_fraction)
432                spectra[:, j, :] = fill_missing(spectra[:,j,:], fd, pulse_times,
433                    mask_replace, valid_rows, noise, interpolate, fill_value)
434            hits[iblock, j, :] = fftshift(np.mean(mask_replace, axis=0))
435            means[iblock, j] = 1 / λ if λ > 0.0 else 0.0
436        # skip inverse FFTs and assignment if not required.
437        if not detect_only:
438            # range inverse STFT
439            z_block[...] = ifft(spectra, axis=2)
440            # sum COLA range blocks into output buffer
441            zout[rows, ...] = 0
442            for j, (cols, window) in enumerate(slices_windows):
443                nw = len(window)
444                zout[rows, cols] += z_block[:, j, :nw]
445
446    return block_times, block_ranges, f, means, isr, hits

12def exp_from_quantile(p, vp, bw=1.0):
13    """
14    Determine the rate parameter of the lifted exponential distribution from a
15    quantile and its corresponding value.
16
17    Parameters
18    ----------
19    p : float | np.ndarray
20        Quantile in [1-bw, 1).  For example, 0.5 for the median.
21    vp : float | np.ndarray
22        Value corresponding to the given quantile.  For example, the median
23        value.
24    bw : float, optional
25        The portion of the distribution governed by an exponential distribution.
26        In the context of the tone-rank algorithm, it's the chirp bandwidth
27        normalized by the sample rate.  Values in interval (0, 1].
28        See notes below.
29
30    Returns
31    -------
32    λ : float | np.ndarray
33        The rate parameter of the lifted exponential distribution.
34
35    Notes
36    -----
37    The definition of the "lifted" exponential distribution is as follows:
38        cdf(x) = (1 - bw) * u(x - x0) + bw * (1 - exp(-λ * x))
39    where u(x) is the unit step function, x0 is some small value (e.g., the
40    noise floor), and other parameters are as described above.
41
42    The idea here is that spectral power in the chirp band should follow an
43    exponential distribution, while the values outside the chirp band will be
44    smaller and follow some other distribution whose shape we don't care about
45    and just model with a step function.
46
47    The formula is the same to solve for a value given the rate parameter.
48    That is, you can also use this function to solve
49        vp = exp_from_quantile(p, λ, bw)
50    """
51    p = np.asarray(p)
52    if not np.all(p >= (1.0 - bw)):
53        raise ValueError("invalid quantile for lifted exponential")
54    vp = np.asarray(vp)

Parameters

For example, the median value. bw : float, optional The portion of the distribution governed by an exponential distribution. In the context of the tone-rank algorithm, it's the chirp bandwidth normalized by the sample rate. Values in interval (0, 1]. See notes below.

Returns

λ : float | np.ndarray The rate parameter of the lifted exponential distribution.

Notes

The definition of the "lifted" exponential distribution is as follows: cdf(x) = (1 - bw) * u(x - x0) + bw * (1 - exp(-λ * x)) where u(x) is the unit step function, x0 is some small value (e.g., the noise floor), and other parameters are as described above.

The idea here is that spectral power in the chirp band should follow an exponential distribution, while the values outside the chirp band will be smaller and follow some other distribution whose shape we don't care about and just model with a step function.

The formula is the same to solve for a value given the rate parameter. That is, you can also use this function to solve vp = exp_from_quantile(p, λ, bw)

 72    loud compared to the expected distribution of the signal. The method assumes
 73    that the power spectrum follows a lifted exponential distribution and uses

 75
 76    Parameters
 77    ----------
 78    spectra : np.ndarray
 79        Complex spectral data, arbitrary shape.
 80    reference_quantile : float, optional
 81        Quantile (in [1-bandwidth, 1)) used to estimate the rate parameter
 82        of the lifted exponential distribution. Default is 0.5 (median).
 83    nominal_false_positive_rate : float, optional

 85        Lower values result in more conservative detection (fewer false alarms
 86        but potentially missed RFI).
 87    bandwidth : float, optional

 89        distribution (as opposed to being noise, filter roll-off, etc).
 90
 91    Returns

 93    mask : np.ndarray[bool]

 95        True indicates RFI detection (sample exceeds threshold).
 96    isr : float
 97        Interference-to-signal ratio: ratio of total power in detected RFI
 98        samples to total power in clean samples.
 99    λ : float
100        Estimated rate parameter of the lifted exponential distribution.
101
102    Notes
103    -----
104    The detection threshold is determined by inverting the cumulative
105    distribution function of the lifted exponential model at the desired
106    false positive rate. See exp_from_quantile for details on the
107    statistical model.
108    """
109    n = np.prod(spectra.shape)
110    power_spectra = abs2(spectra)
111    power_spectra.shape = (n,)
112    # partition seems to be a little faster than quantile, I guess because the
113    # latter does a full sort and interpolates?  For large n the difference is
114    # small.
115    k = round(reference_quantile * n)
116    kth_power = np.partition(power_spectra, k)[k]
117    if kth_power == 0.0:
118        return np.zeros(spectra.shape, dtype=bool), 0.0, 0.0
119    # Estimate the parameter of a lifted exponential distribution.
120    λ = exp_from_quantile(reference_quantile, kth_power, bandwidth)
121    # Determine threshold.  Since we've estimated a statistical model, we can
122    # achieve a specified nominal false positive rate by thresholding the
123    # nominal CDF at one minus that value.
124    q = 1.0 - nominal_false_positive_rate
125    too_loud = exp_from_quantile(q, λ, bandwidth)
126    # Use threshold to generate mask.
127    mask = power_spectra >= too_loud
128    # Compute interference-to-signal ratio now that we've labeled the data.
129    # Since masked values are presumably the sum of signal and interference,
130    # we'll include an arithmetic correction using the stats model.
131    mean_power = 1 / λ
132    masked_sig_power = np.sum(mask) * mean_power
133    rfi_power = np.sum(power_spectra[mask]) - masked_sig_power
134    sig_power = np.sum(power_spectra[~mask]) + masked_sig_power
135    isr = rfi_power / sig_power
136    # Unflatten data.
137    mask.shape = spectra.shape
138    return mask, isr, λ

141def fill_missing(z, fd, t, mask_replace, valid_rows, noise, interpolate=True,
142                 fill_value="noise"):
143    """
144    Fill missing/RFI-contaminated samples in spectral domain data.
145
146    Parameters
147    ----------
148    z : np.ndarray
149        Complex spectral data to fill, shape (m, n)
150    fd : float
151        Doppler centroid frequency in Hz.  It is not scaled for each frequency
152        bin in `z`.  If wideband processing is needed, just baseband `z` before
153        calling this function and set `fd=0`.
154    t : np.ndarray
155        Pulse times in seconds since epoch, length m
156    mask_replace : np.ndarray
157        Boolean mask marking samples to replace, shape (m, n)
158    valid_rows : np.ndarray
159        Boolean mask marking valid rows, shape (m,)
160    noise : np.ndarray
161        Random noise samples in 1D array for fallback replacement.  Values may
162        be used multiple times if length is less numpy.prod((m, n)).
163    interpolate : bool, optional
164        If True, attempt linear/nearest-neighbor interpolation. If False,
165        replace directly with fill_value. Default is True.
166    fill_value : str, optional
167        Fallback strategy when interpolation fails or is disabled.
168        "noise": use provided noise vector (default)
169        "zero": use zero
170
171    Returns
172    -------
173    zout : np.ndarray
174        Filled data, same shape as z
175    """
176    m, n = z.shape
177    if len(t) != m:
178        raise ValueError(f"expected len(t)=={m} got {len(t)}")
179    if mask_replace.shape != (m, n):
180        raise ValueError(f"{mask_replace.shape=} does not match {z.shape=}")
181    if valid_rows.shape != (m,):
182        raise ValueError(f"{valid_rows.shape=} does not match {z.shape[0]=}")
183    if interpolate and m < 2:
184        log.warning(f"Disabling interpolation for block of {m} rows "
185            "because there are no neighbors to use for interpolation.")
186        interpolate = False
187
188    # Deramp Doppler.
189    deramp = np.exp(-1j * 2 * np.pi * fd * t).astype(z.dtype)
190    zout = deramp[:, None] * z
191
192    # Exclude RFI samples from being used as interpolation sources.
193    mask_valid_clean = valid_rows[:, None] & ~mask_replace
194
195    for i in range(m):
196        # copy for modification
197        cols_need_replacement = mask_replace[i, :].copy()
198
199        if interpolate:
200            # Previous and next pulse, with reflection boundary condition.
201            iprev, inext = i - 1, i + 1
202            if i == 0:
203                iprev = i + 1
204            if i == m - 1:
205                inext = i - 1
206
207            # Non-uniform time sampling, so let's weight closer samples more.
208            # NOTE abs() since we might've reflected.
209            dt_prev = abs(t[i] - t[iprev])
210            dt_next = abs(t[inext] - t[i])
211            w_prev = dt_next / (dt_prev + dt_next)
212            w_next = dt_prev / (dt_prev + dt_next)
213
214            # Four cases for replacement:
215            # 1. prev and next both valid -> lerp between them
216            j = np.where(mask_valid_clean[iprev, :] & mask_valid_clean[inext, :]
217                & cols_need_replacement)[0]
218            zout[i, j] = w_prev * zout[iprev, j] + w_next * zout[inext, j]
219            cols_need_replacement[j] = False
220
221            # 2. only prev valid. use it
222            j = np.where(mask_valid_clean[iprev, :] & cols_need_replacement)[0]
223            zout[i, j] = zout[iprev, j]
224            cols_need_replacement[j] = False
225
226            # 3. only next valid. use it
227            j = np.where(mask_valid_clean[inext, :] & cols_need_replacement)[0]
228            zout[i, j] = zout[inext, j]
229            cols_need_replacement[j] = False
230
231        # 4. Fallback for remaining samples (or all if not interpolating)
232        j = np.where(cols_need_replacement)[0]
233        if len(j) > 0:
234            if fill_value == "zero":
235                zout[i, j] = 0.0
236            elif fill_value == "noise":
237                noise_idx = ((i * n) + j) % len(noise)
238                zout[i, j] = noise[noise_idx]
239            else:
240                raise ValueError(f"Invalid fill_value: {fill_value}")
241
242    # Put Doppler back on.
243    zout *= deramp[:, None].conj()
244    return zout

Creates datasets for time/range/frequency axes and detection results (signal mean, interference-to-signal ratio, and hit counts) with appropriate metadata attributes.

Parameters

group : h5py.Group HDF5 group to write datasets into. t : np.ndarray[float] Azimuth times at center of each azimuth block in seconds since epoch, length num_az_blocks. Written as "nativeDopplerTime" dataset. r : np.ndarray[float] Slant ranges at center of each range block in meters, length num_range_blocks. Written as "slantRange" dataset. freq : np.ndarray[float] Frequency axis in Hz, length num_freq_bins. Written as "frequency" dataset. means : np.ndarray[float] Estimated mean signal power per block from lifted exponential model, shape (num_az_blocks, num_range_blocks). Written as "signalMean" dataset. isr : np.ndarray[float] Interference-to-signal ratio per block, shape (num_az_blocks, num_range_blocks). Written as "interferenceSignalRatio" dataset. hits : np.ndarray[float] Fraction of pulses with detected RFI at each frequency bin, shape (num_az_blocks, num_range_blocks, num_freq_bins). Written as "hitCount" dataset.
